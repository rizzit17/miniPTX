cmake_minimum_required(VERSION 3.16)
project(MiniPTX VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(miniptx_core src/lexer.cpp src/parser.cpp src/semantic.cpp src/ir.cpp src/codegen.cpp src/compiler.cpp)
target_include_directories(miniptx_core PUBLIC include)
if(MSVC)
  target_compile_options(miniptx_core PRIVATE /W4)
else()
  target_compile_options(miniptx_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(miniptx src/main.cpp)
target_link_libraries(miniptx PRIVATE miniptx_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(compiler_tests tests/compiler_tests.cpp)
  target_link_libraries(compiler_tests PRIVATE miniptx_core)
  add_test(NAME compiler_tests COMMAND compiler_tests)
endif()
